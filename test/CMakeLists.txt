cmake_minimum_required(VERSION 3.14)
project(p101_numeric_tests C CXX)
enable_testing()

set(P101_API_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/../api-manifest.tsv")
set(P101_UNIT_TEST_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/unit-test-manifest.tsv")
if(NOT EXISTS "${P101_API_MANIFEST}" OR NOT EXISTS "${P101_UNIT_TEST_MANIFEST}")
    message(FATAL_ERROR "Wrapper API and unit-test manifests are required")
endif()
file(STRINGS "${P101_API_MANIFEST}" P101_API_ROWS REGEX "^p101_")
file(STRINGS "${P101_UNIT_TEST_MANIFEST}" P101_UNIT_ROWS REGEX "^p101_")
set(P101_API_WRAPPERS "")
set(P101_UNIT_WRAPPERS "")
foreach(row IN LISTS P101_API_ROWS)
    string(REGEX MATCH "^p101_[A-Za-z0-9_]+" wrapper "${row}")
    list(APPEND P101_API_WRAPPERS "${wrapper}")
endforeach()
foreach(row IN LISTS P101_UNIT_ROWS)
    string(REGEX MATCH "^p101_[A-Za-z0-9_]+" wrapper "${row}")
    list(APPEND P101_UNIT_WRAPPERS "${wrapper}")
endforeach()
set(P101_UNIT_WRAPPERS_UNIQUE ${P101_UNIT_WRAPPERS})
list(REMOVE_DUPLICATES P101_UNIT_WRAPPERS_UNIQUE)
list(LENGTH P101_API_WRAPPERS P101_API_COUNT)
list(LENGTH P101_UNIT_WRAPPERS P101_UNIT_COUNT)
list(LENGTH P101_UNIT_WRAPPERS_UNIQUE P101_UNIT_UNIQUE_COUNT)
if(NOT P101_API_COUNT EQUAL P101_UNIT_COUNT OR NOT P101_UNIT_COUNT EQUAL P101_UNIT_UNIQUE_COUNT)
    message(FATAL_ERROR "Every public wrapper must have exactly one unit-test row")
endif()

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(P101_PUBLIC_INCLUDE_DIRS "" CACHE STRING "Extra p101 include dirs")
set(P101_PUBLIC_LINK_DIRS "" CACHE STRING "Extra p101 link dirs")
separate_arguments(P101_PUBLIC_INCLUDE_DIRS_LIST NATIVE_COMMAND "${P101_PUBLIC_INCLUDE_DIRS}")
separate_arguments(P101_PUBLIC_LINK_DIRS_LIST NATIVE_COMMAND "${P101_PUBLIC_LINK_DIRS}")

foreach(library IN ITEMS MATH RANDOM ENV ERROR)
    string(TOLOWER "${library}" library_name)
    find_library(P101_${library}_LIBRARY NAMES p101_${library_name} PATHS ${P101_PUBLIC_LINK_DIRS_LIST} NO_DEFAULT_PATH)
    if(NOT P101_${library}_LIBRARY)
        find_library(P101_${library}_LIBRARY NAMES p101_${library_name})
    endif()
    if(NOT P101_${library}_LIBRARY)
        message(FATAL_ERROR "Could not find p101_${library_name}")
    endif()
endforeach()

set(P101_TEST_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/../include" ${P101_PUBLIC_INCLUDE_DIRS_LIST})
set(P101_TEST_LIBRARIES "${P101_RANDOM_LIBRARY}" "${P101_MATH_LIBRARY}" "${P101_ENV_LIBRARY}" "${P101_ERROR_LIBRARY}")

include(${CMAKE_CURRENT_SOURCE_DIR}/fault_shards.cmake)
foreach(test_name IN LISTS P101_FAULT_SHARD_TESTS ITEMS test_library test_headers_c)
    add_executable(${test_name} ${test_name}.c)
    target_include_directories(${test_name} PRIVATE ${P101_TEST_INCLUDE_DIRS})
    target_compile_definitions(${test_name} PRIVATE _POSIX_C_SOURCE=200809L _XOPEN_SOURCE=700)
    target_link_libraries(${test_name} PRIVATE ${P101_TEST_LIBRARIES})
    add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()

add_executable(test_behavior random/test_behavior.c)
target_include_directories(test_behavior PRIVATE ${P101_TEST_INCLUDE_DIRS})
target_compile_definitions(test_behavior PRIVATE _POSIX_C_SOURCE=200809L _XOPEN_SOURCE=700)
target_link_libraries(test_behavior PRIVATE ${P101_TEST_LIBRARIES})
add_test(NAME test_behavior COMMAND test_behavior)

add_executable(test_headers_cxx test_headers.cpp)
target_include_directories(test_headers_cxx PRIVATE ${P101_TEST_INCLUDE_DIRS})
target_compile_definitions(test_headers_cxx PRIVATE _POSIX_C_SOURCE=200809L _XOPEN_SOURCE=700)
target_link_libraries(test_headers_cxx PRIVATE ${P101_TEST_LIBRARIES})
add_test(NAME test_headers_cxx COMMAND test_headers_cxx)

if(APPLE)
    set(P101_PLATFORM_DEFINITIONS _DARWIN_C_SOURCE)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(P101_PLATFORM_DEFINITIONS _GNU_SOURCE)
elseif(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
    set(P101_PLATFORM_DEFINITIONS _BSD_SOURCE __BSD_VISIBLE)
endif()
foreach(test_name IN LISTS P101_FAULT_SHARD_TESTS ITEMS test_library test_headers_c test_headers_cxx test_behavior)
    target_compile_definitions(${test_name} PRIVATE ${P101_PLATFORM_DEFINITIONS})
endforeach()

if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    foreach(test_name IN LISTS P101_FAULT_SHARD_TESTS ITEMS test_library test_headers_c test_behavior)
        target_compile_options(${test_name} PRIVATE -Wall -Wextra -Werror -pedantic-errors)
    endforeach()
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(test_headers_cxx PRIVATE -Wall -Wextra -Werror -pedantic-errors)
endif()
